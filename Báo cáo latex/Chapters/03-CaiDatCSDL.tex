\chapter{CÀI ĐẶT CƠ SỞ DỮ LIỆU, BỘ KÍCH HOẠT VÀ THỦ TỤC GIAO DỊCH}

\section{Môi trường cài đặt và cấu trúc bảng dữ liệu}

Hệ thống cơ sở dữ liệu được cài đặt trên hệ quản trị PostgreSQL. Toàn bộ 11 bảng dữ liệu sau khi chuẩn hóa BCNF ở Chương 2 được tạo lập với đầy đủ khóa chính, khóa ứng viên, khóa ngoại, các ràng buộc miền giá trị và chỉ mục B-Tree (B-Tree Index) phục vụ tối ưu hóa truy vấn.

\begin{lstlisting}[language=SQL, caption={Kịch bản tạo 11 bảng dữ liệu, ràng buộc toàn vẹn và chỉ mục B-Tree}]
-- 1. BANG SAN VAN DONG (Phuc vu phan manh theo KhuVuc)
CREATE TABLE SAN_VAN_DONG (
    MaSan VARCHAR(10) PRIMARY KEY,
    TenSan VARCHAR(100) NOT NULL UNIQUE,
    DiaChi VARCHAR(200) NOT NULL,
    SucChua INT NOT NULL CHECK (SucChua > 0),
    KhuVuc VARCHAR(20) NOT NULL DEFAULT 'MienBac' CHECK (KhuVuc IN ('MienBac', 'MienNam'))
);

-- 2. BANG DOI BONG
CREATE TABLE DOI_BONG (
    MaDoi VARCHAR(10) PRIMARY KEY,
    TenDoi VARCHAR(100) NOT NULL UNIQUE,
    HLV VARCHAR(100) NOT NULL,
    MaSanNha VARCHAR(10) NOT NULL REFERENCES SAN_VAN_DONG(MaSan) ON UPDATE CASCADE
);

-- 3. BANG MAU AO DOI BONG
CREATE TABLE MAU_AO_DOI_BONG (
    MaDoi VARCHAR(10) NOT NULL REFERENCES DOI_BONG(MaDoi) ON DELETE CASCADE ON UPDATE CASCADE,
    LoaiMauAo VARCHAR(20) NOT NULL CHECK (LoaiMauAo IN ('SanNha', 'SanKhach', 'Phu')),
    MoTaMau VARCHAR(50) NOT NULL,
    PRIMARY KEY (MaDoi, LoaiMauAo)
);

-- 4. BANG CAU THU
CREATE TABLE CAU_THU (
    MaCauThu VARCHAR(10) PRIMARY KEY,
    HoTen VARCHAR(100) NOT NULL,
    NgaySinh DATE NOT NULL,
    ViTri VARCHAR(20) NOT NULL CHECK (ViTri IN ('ThuMon', 'HauVe', 'TienVe', 'TienDao')),
    QuocTich VARCHAR(50) NOT NULL DEFAULT 'Viet Nam'
);

-- 5. BANG HOP DONG CAU THU (Luu lich su, ap dung Partial Unique Index)
CREATE TABLE HOP_DONG_CAU_THU (
    MaHD VARCHAR(10) PRIMARY KEY,
    MaCauThu VARCHAR(10) NOT NULL REFERENCES CAU_THU(MaCauThu) ON UPDATE CASCADE,
    MaDoiChuQuan VARCHAR(10) NOT NULL REFERENCES DOI_BONG(MaDoi) ON UPDATE CASCADE,
    MaDoiHienTai VARCHAR(10) NOT NULL REFERENCES DOI_BONG(MaDoi) ON UPDATE CASCADE,
    SoAo INT NOT NULL CHECK (SoAo BETWEEN 1 AND 35),
    NgayBatDau DATE NOT NULL,
    NgayKetThuc DATE NOT NULL,
    TrangThai VARCHAR(20) NOT NULL CHECK (TrangThai IN ('ChinhThuc', 'ChoMuon', 'TamDung', 'ThanhLy')),
    CHECK (NgayKetThuc >= NgayBatDau)
);

-- Partial Unique Indexes cho cac hop dong dang hieu luc thi dau:
CREATE UNIQUE INDEX idx_uq_cauthu_active_contract 
ON HOP_DONG_CAU_THU (MaCauThu) WHERE TrangThai IN ('ChinhThuc', 'ChoMuon');

CREATE UNIQUE INDEX idx_uq_doihientai_soao_active 
ON HOP_DONG_CAU_THU (MaDoiHienTai, SoAo) WHERE TrangThai IN ('ChinhThuc', 'ChoMuon');

-- 6. BANG TRONG TAI
CREATE TABLE TRONG_TAI (
    MaTrongTai VARCHAR(10) PRIMARY KEY,
    HoTen VARCHAR(100) NOT NULL,
    NgaySinh DATE NOT NULL,
    QueQuan VARCHAR(100) NOT NULL,
    CapBac VARCHAR(50) NOT NULL DEFAULT 'Quoc gia'
);

-- 7. BANG VONG DAU
CREATE TABLE VONG_DAU (
    MaVong INT PRIMARY KEY CHECK (MaVong BETWEEN 1 AND 10),
    TenVong VARCHAR(50) NOT NULL UNIQUE,
    ThuTu INT NOT NULL UNIQUE CHECK (ThuTu BETWEEN 1 AND 10)
);

-- 8. BANG TRAN DAU
CREATE TABLE TRAN_DAU (
    MaTran VARCHAR(10) PRIMARY KEY,
    MaVong INT NOT NULL REFERENCES VONG_DAU(MaVong) ON UPDATE CASCADE,
    MaDoiNha VARCHAR(10) NOT NULL REFERENCES DOI_BONG(MaDoi) ON UPDATE CASCADE,
    MaDoiKhach VARCHAR(10) NOT NULL REFERENCES DOI_BONG(MaDoi) ON UPDATE CASCADE,
    MaSan VARCHAR(10) NOT NULL REFERENCES SAN_VAN_DONG(MaSan) ON UPDATE CASCADE,
    NgayGio TIMESTAMP NOT NULL,
    GiaVe DECIMAL(12, 2) NOT NULL DEFAULT 50000 CHECK (GiaVe >= 0),
    SoKhach INT DEFAULT 0 CHECK (SoKhach >= 0),
    BanThangNha INT DEFAULT 0 CHECK (BanThangNha >= 0),
    BanThangKhach INT DEFAULT 0 CHECK (BanThangKhach >= 0),
    TheVangNha INT DEFAULT 0 CHECK (TheVangNha >= 0),
    TheVangKhach INT DEFAULT 0 CHECK (TheVangKhach >= 0),
    TheDoNha INT DEFAULT 0 CHECK (TheDoNha >= 0),
    TheDoKhach INT DEFAULT 0 CHECK (TheDoKhach >= 0),
    TrangThai VARCHAR(20) NOT NULL DEFAULT 'ChuaDienRa' CHECK (TrangThai IN ('ChuaDienRa', 'DaKetThuc')),
    CHECK (MaDoiNha <> MaDoiKhach),
    UNIQUE (MaDoiNha, MaDoiKhach),
    UNIQUE (MaVong, MaDoiNha),
    UNIQUE (MaVong, MaDoiKhach)
);

-- 9. BANG PHAN CONG TRONG TAI
CREATE TABLE PHAN_CONG_TRONG_TAI (
    MaTran VARCHAR(10) NOT NULL REFERENCES TRAN_DAU(MaTran) ON DELETE CASCADE ON UPDATE CASCADE,
    VaiTro VARCHAR(20) NOT NULL CHECK (VaiTro IN ('Chinh', 'Bien1', 'Bien2', 'Ban')),
    MaTrongTai VARCHAR(10) NOT NULL REFERENCES TRONG_TAI(MaTrongTai) ON UPDATE CASCADE,
    PRIMARY KEY (MaTran, VaiTro),
    UNIQUE (MaTran, MaTrongTai)
);

-- 10. BANG DANG KY THI DAU
CREATE TABLE DANG_KY_THI_DAU (
    MaTran VARCHAR(10) NOT NULL REFERENCES TRAN_DAU(MaTran) ON DELETE CASCADE ON UPDATE CASCADE,
    MaCauThu VARCHAR(10) NOT NULL REFERENCES CAU_THU(MaCauThu) ON UPDATE CASCADE,
    MaDoi VARCHAR(10) NOT NULL REFERENCES DOI_BONG(MaDoi) ON UPDATE CASCADE,
    VaiTroThiDau VARCHAR(20) NOT NULL CHECK (VaiTroThiDau IN ('ChinhThuc', 'DuBi')),
    SoAoDangKy INT NOT NULL CHECK (SoAoDangKy BETWEEN 1 AND 35),
    PRIMARY KEY (MaTran, MaCauThu),
    UNIQUE (MaTran, MaDoi, SoAoDangKy)
);

-- 11. BANG SU KIEN TRAN DAU (Sau chuan hoa BCNF: Khoa ngoai to hop sang DANG_KY_THI_DAU)
CREATE TABLE SU_KIEN_TRAN_DAU (
    MaSuKien VARCHAR(15) PRIMARY KEY,
    MaTran VARCHAR(10) NOT NULL,
    MaCauThu VARCHAR(10) NOT NULL,
    LoaiSuKien VARCHAR(20) NOT NULL CHECK (LoaiSuKien IN ('BanThang', 'PhanLuoi', 'TheVang', 'TheDo')),
    PhutGhiNhan INT NOT NULL CHECK (PhutGhiNhan BETWEEN 1 AND 120),
    FOREIGN KEY (MaTran, MaCauThu) REFERENCES DANG_KY_THI_DAU(MaTran, MaCauThu) ON DELETE CASCADE ON UPDATE CASCADE
);

-- CHI MUC B-TREE TOI UU HOA TRUY VAN
CREATE INDEX idx_tran_dau_san_ngaygio ON TRAN_DAU(MaSan, NgayGio);
CREATE INDEX idx_tran_dau_vong ON TRAN_DAU(MaVong);
CREATE INDEX idx_dangky_tran_doi ON DANG_KY_THI_DAU(MaTran, MaDoi);
CREATE INDEX idx_sukien_tran_cauthu ON SU_KIEN_TRAN_DAU(MaTran, MaCauThu);
\end{lstlisting}

\section{Cài đặt năm bộ kích hoạt kiểm tra toàn vẹn nghiệp vụ và điều khiển tương tranh}

Trong cơ chế kiểm soát tương tranh đa phiên bản (MVCC -- Multi-Version Concurrency Control) của PostgreSQL với mức cô lập mặc định \texttt{READ COMMITTED}, nếu một bộ kích hoạt chỉ thực hiện câu lệnh đếm thông thường (\texttt{SELECT COUNT(*)}) mà không khóa bản ghi cha tương ứng, hai giao dịch chạy đồng thời có thể cùng đọc trạng thái cũ và cùng vượt qua bước kiểm tra (dẫn đến điều kiện tranh chấp -- Race Condition hoặc mất mát dữ liệu cập nhật -- Lost Update). Để khắc phục triệt để, các bộ kích hoạt dưới đây sử dụng kỹ thuật khóa dòng (Row-level lock \texttt{FOR NO KEY UPDATE}) trên thực thể cha (\texttt{SAN\_VAN\_DONG}, \texttt{DOI\_BONG}, \texttt{TRAN\_DAU}) nhằm tuần tự hóa các giao dịch ghi cạnh tranh trên cùng một thực thể nghiệp vụ.

\subsection{Bộ kích hoạt kiểm tra quy định sân đấu và thời gian trận đấu}
Đảm bảo số lượng khán giả không vượt quá sức chứa sân, thời điểm kết thúc hợp lệ và hai trận đấu trên cùng một sân cách nhau ít nhất 4 giờ:

\begin{lstlisting}[language=SQL, caption={Bộ kích hoạt kiểm tra sức chứa và trùng lịch sân vận động}]
CREATE OR REPLACE FUNCTION fn_check_quy_dinh_san_dau()
RETURNS TRIGGER AS $$
DECLARE
    v_suc_chua INT;
    v_so_tran_trung_gio INT;
BEGIN
    -- Khoa dong san van dong cha de tuan tu hoa giao dich xep lich
    SELECT SucChua INTO v_suc_chua
    FROM SAN_VAN_DONG WHERE MaSan = NEW.MaSan FOR NO KEY UPDATE;

    IF NEW.SoKhach > v_suc_chua THEN
        RAISE EXCEPTION 'Loi nghiep vu: So luong khan gia (%) vuot qua suc chua san (%)!',
            NEW.SoKhach, v_suc_chua;
    END IF;

    IF NEW.TrangThai = 'DaKetThuc' AND NEW.NgayGio > CURRENT_TIMESTAMP THEN
        RAISE EXCEPTION 'Loi nghiep vu: Tran dau o thoi diem tuong lai khong the DaKetThuc!';
    END IF;

    -- Kiem tra cach nhau 4 gio (khai thac chi muc B-Tree idx_tran_dau_san_ngaygio)
    SELECT COUNT(*) INTO v_so_tran_trung_gio
    FROM TRAN_DAU
    WHERE MaSan = NEW.MaSan AND MaTran <> NEW.MaTran
      AND NgayGio > NEW.NgayGio - INTERVAL '4 hours'
      AND NgayGio < NEW.NgayGio + INTERVAL '4 hours';

    IF v_so_tran_trung_gio > 0 THEN
        RAISE EXCEPTION 'Loi nghiep vu: San (%) da co tran dau khac trong khoang 4 gio!', NEW.MaSan;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_check_quy_dinh_san_dau ON TRAN_DAU;
CREATE TRIGGER trg_check_quy_dinh_san_dau
BEFORE INSERT OR UPDATE OF MaSan, NgayGio, SoKhach, TrangThai ON TRAN_DAU
FOR EACH ROW EXECUTE FUNCTION fn_check_quy_dinh_san_dau();
\end{lstlisting}

\subsection{Bộ kích hoạt kiểm tra hợp đồng, số áo duy nhất và sĩ số tối đa 30 cầu thủ}
Đảm bảo mỗi cầu thủ chỉ thi đấu cho một đội bóng tại một thời điểm, không trùng số áo trong đội và mỗi đội đăng ký tối đa 30 cầu thủ đang có hợp đồng hiệu lực:

\begin{lstlisting}[language=SQL, caption={Bộ kích hoạt kiểm tra hợp đồng, số áo và giới hạn 30 cầu thủ}]
CREATE OR REPLACE FUNCTION fn_check_hop_dong_va_so_ao()
RETURNS TRIGGER AS $$
DECLARE
    v_so_hop_dong_hieu_luc INT;
    v_trung_so_ao INT;
    v_so_luong_cau_thu INT;
BEGIN
    PERFORM 1 FROM DOI_BONG WHERE MaDoi = NEW.MaDoiHienTai FOR NO KEY UPDATE;

    IF NEW.TrangThai IN ('ChinhThuc', 'ChoMuon') THEN
        SELECT COUNT(*) INTO v_so_hop_dong_hieu_luc
        FROM HOP_DONG_CAU_THU
        WHERE MaCauThu = NEW.MaCauThu AND MaHD <> NEW.MaHD
          AND TrangThai IN ('ChinhThuc', 'ChoMuon')
          AND NOT (NgayKetThuc < NEW.NgayBatDau OR NgayBatDau > NEW.NgayKetThuc);

        IF v_so_hop_dong_hieu_luc > 0 THEN
            RAISE EXCEPTION 'Loi nghiep vu: Cau thu (%) dang co hop dong hieu luc khac!', NEW.MaCauThu;
        END IF;

        SELECT COUNT(*) INTO v_trung_so_ao
        FROM HOP_DONG_CAU_THU
        WHERE MaDoiHienTai = NEW.MaDoiHienTai AND SoAo = NEW.SoAo AND MaHD <> NEW.MaHD
          AND TrangThai IN ('ChinhThuc', 'ChoMuon')
          AND NOT (NgayKetThuc < NEW.NgayBatDau OR NgayBatDau > NEW.NgayKetThuc);

        IF v_trung_so_ao > 0 THEN
            RAISE EXCEPTION 'Loi nghiep vu: So ao (%) da duoc su dung tai doi (%)!', NEW.SoAo, NEW.MaDoiHienTai;
        END IF;

        SELECT COUNT(*) INTO v_so_luong_cau_thu
        FROM HOP_DONG_CAU_THU
        WHERE MaDoiHienTai = NEW.MaDoiHienTai AND MaHD <> NEW.MaHD
          AND TrangThai IN ('ChinhThuc', 'ChoMuon');

        IF v_so_luong_cau_thu >= 30 THEN
            RAISE EXCEPTION 'Loi nghiep vu: Doi bong (%) da dat toi da 30 cau thu!', NEW.MaDoiHienTai;
        END IF;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_check_hop_dong_va_so_ao ON HOP_DONG_CAU_THU;
CREATE TRIGGER trg_check_hop_dong_va_so_ao
BEFORE INSERT OR UPDATE OF MaDoiHienTai, SoAo, NgayBatDau, NgayKetThuc, TrangThai ON HOP_DONG_CAU_THU
FOR EACH ROW EXECUTE FUNCTION fn_check_hop_dong_va_so_ao();
\end{lstlisting}

\subsection{Bộ kích hoạt kiểm tra đăng ký thi đấu (11 đá chính, tối đa 7 dự bị) và treo giò}
Kiểm tra biên chế hợp lệ theo ngày diễn ra trận đấu, giới hạn 11 cầu thủ đá chính, tối đa 7 cầu thủ dự bị (tổng tối đa 18 cầu thủ mỗi đội mỗi trận theo công thức $11 + 7 = 18$), điều khoản cho mượn và án treo giò (do nhận thẻ đỏ trực tiếp hoặc tích lũy đủ 2 thẻ vàng ở trận liền trước):

\begin{lstlisting}[language=SQL, caption={Bộ kích hoạt kiểm tra đăng ký thi đấu và kiểm soát treo giò}]
CREATE OR REPLACE FUNCTION fn_check_dang_ky_thi_dau_va_treo_gio()
RETURNS TRIGGER AS $$
DECLARE
    v_ma_vong INT; v_doi_nha VARCHAR(10); v_doi_khach VARCHAR(10);
    v_ngay_gio TIMESTAMP; v_doi_chu_quan VARCHAR(10); v_doi_hien_tai VARCHAR(10);
    v_so_chinh_thuc INT; v_so_du_bi INT; v_tong_dang_ky INT;
    v_the_do_gan_nhat INT; v_the_vang_gan_nhat INT;
    v_tong_the_vang_tich_luy INT; v_ma_tran_truoc VARCHAR(10);
BEGIN
    SELECT MaVong, MaDoiNha, MaDoiKhach, NgayGio 
    INTO v_ma_vong, v_doi_nha, v_doi_khach, v_ngay_gio
    FROM TRAN_DAU WHERE MaTran = NEW.MaTran FOR NO KEY UPDATE;

    IF NEW.MaDoi NOT IN (v_doi_nha, v_doi_khach) THEN
        RAISE EXCEPTION 'Loi nghiep vu: Doi (%) khong tham gia tran (%)!', NEW.MaDoi, NEW.MaTran;
    END IF;

    -- Kiem tra hop dong cau thu con hieu luc tai dung ngay dien ra tran dau
    SELECT MaDoiChuQuan, MaDoiHienTai INTO v_doi_chu_quan, v_doi_hien_tai
    FROM HOP_DONG_CAU_THU
    WHERE MaCauThu = NEW.MaCauThu AND MaDoiHienTai = NEW.MaDoi
      AND TrangThai IN ('ChinhThuc', 'ChoMuon')
      AND v_ngay_gio::DATE BETWEEN NgayBatDau AND NgayKetThuc LIMIT 1;

    IF v_doi_hien_tai IS NULL THEN
        RAISE EXCEPTION 'Loi nghiep vu: Cau thu (%) khong co hop dong hieu luc tai doi (%) vao ngay thi dau!', 
            NEW.MaCauThu, NEW.MaDoi;
    END IF;

    IF v_doi_chu_quan <> v_doi_hien_tai AND v_doi_chu_quan IN (v_doi_nha, v_doi_khach) THEN
        RAISE EXCEPTION 'Loi nghiep vu: Cau thu cho muon khong duoc thi dau chong lai doi chu quan!';
    END IF;

    -- Kiem tra gioi han danh sach dang ky: 11 da chinh, toi da 7 du bi, tong toi da 18
    SELECT COUNT(*) FILTER (WHERE VaiTroThiDau = 'ChinhThuc'),
           COUNT(*) FILTER (WHERE VaiTroThiDau = 'DuBi'),
           COUNT(*)
    INTO v_so_chinh_thuc, v_so_du_bi, v_tong_dang_ky
    FROM DANG_KY_THI_DAU
    WHERE MaTran = NEW.MaTran AND MaDoi = NEW.MaDoi AND MaCauThu <> NEW.MaCauThu;

    IF v_tong_dang_ky >= 18 THEN
        RAISE EXCEPTION 'Loi nghiep vu: Doi (%) da dang ky du toi da 18 cau thu!', NEW.MaDoi;
    END IF;
    IF NEW.VaiTroThiDau = 'ChinhThuc' AND v_so_chinh_thuc >= 11 THEN
        RAISE EXCEPTION 'Loi nghiep vu: Doi (%) da du 11 cau thu da chinh!', NEW.MaDoi;
    END IF;
    IF NEW.VaiTroThiDau = 'DuBi' AND v_so_du_bi >= 7 THEN
        RAISE EXCEPTION 'Loi nghiep vu: Doi (%) da co du toi da 7 cau thu du bi (tong toi da 18)!', NEW.MaDoi;
    END IF;

    -- Kiem tra an treo gio tu tran dau lien truoc cua doi bong
    SELECT MaTran INTO v_ma_tran_truoc
    FROM TRAN_DAU
    WHERE (MaDoiNha = NEW.MaDoi OR MaDoiKhach = NEW.MaDoi)
      AND NgayGio < v_ngay_gio AND TrangThai = 'DaKetThuc'
    ORDER BY NgayGio DESC LIMIT 1;

    IF v_ma_tran_truoc IS NOT NULL THEN
        -- 1. The do truc tiep o tran lien truoc
        SELECT COUNT(*) INTO v_the_do_gan_nhat
        FROM SU_KIEN_TRAN_DAU
        WHERE MaTran = v_ma_tran_truoc AND MaCauThu = NEW.MaCauThu AND LoaiSuKien = 'TheDo';

        -- 2. The vang nhan o tran lien truoc
        SELECT COUNT(*) INTO v_the_vang_gan_nhat
        FROM SU_KIEN_TRAN_DAU
        WHERE MaTran = v_ma_tran_truoc AND MaCauThu = NEW.MaCauThu AND LoaiSuKien = 'TheVang';

        IF v_the_do_gan_nhat > 0 OR v_the_vang_gan_nhat >= 2 THEN
            RAISE EXCEPTION 'Loi nghiep vu: Cau thu (%) bi treo gio do nhan the do o tran lien truoc!', 
                NEW.MaCauThu;
        END IF;

        -- 3. Tich luy the vang: The vang o tran truoc hoan tat du bo 2 the vang (the thu 2, 4...)
        IF v_the_vang_gan_nhat = 1 THEN
            SELECT COUNT(*) INTO v_tong_the_vang_tich_luy
            FROM SU_KIEN_TRAN_DAU sk
            JOIN TRAN_DAU td ON sk.MaTran = td.MaTran
            WHERE sk.MaCauThu = NEW.MaCauThu AND sk.LoaiSuKien = 'TheVang'
              AND td.NgayGio <= (SELECT NgayGio FROM TRAN_DAU WHERE MaTran = v_ma_tran_truoc);

            IF (v_tong_the_vang_tich_luy % 2) = 0 THEN
                RAISE EXCEPTION 'Loi nghiep vu: Cau thu (%) bi treo gio do tich luy du % the vang!',
                    NEW.MaCauThu, v_tong_the_vang_tich_luy;
            END IF;
        END IF;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_check_dang_ky_thi_dau_va_treo_gio ON DANG_KY_THI_DAU;
CREATE TRIGGER trg_check_dang_ky_thi_dau_va_treo_gio
BEFORE INSERT OR UPDATE OF MaTran, MaCauThu, MaDoi, VaiTroThiDau ON DANG_KY_THI_DAU
FOR EACH ROW EXECUTE FUNCTION fn_check_dang_ky_thi_dau_va_treo_gio();
\end{lstlisting}

\textit{Đặc tính bảo toàn chu kỳ kỷ luật}: Thuật toán kiểm tra tính chẵn của tổng thẻ vàng tích lũy (\texttt{v\_tong\_the\_vang\_tich\_luy \% 2 = 0}) kết hợp với điều kiện cầu thủ vừa nhận thẻ vàng ở trận liền trước (\texttt{v\_the\_vang\_gan\_nhat = 1}) bảo đảm tính toàn vẹn: Khi một cầu thủ nhận 2 thẻ vàng trong cùng một trận (tương đương 1 thẻ đỏ gián tiếp), bộ kích hoạt xử lý án treo giò ngay tại nhánh kiểm tra thẻ đỏ, đồng thời 2 thẻ vàng này làm tăng tổng số thẻ vàng thêm 2 (giữ nguyên tính chẵn lẻ của chuỗi tích lũy), nhờ đó chu kỳ tích lũy thẻ vàng đơn lẻ ở các trận tiếp theo không bao giờ bị sai lệch.

\subsection{Bộ kích hoạt giới hạn một trọng tài điều khiển tối đa một trận mỗi vòng}
Sử dụng khóa dòng trên bảng \texttt{TRONG\_TAI} và kiểm tra mỗi trọng tài chỉ xuất hiện tối đa 1 lần trong cùng một vòng đấu:

\begin{lstlisting}[language=SQL, caption={Bộ kích hoạt kiểm tra phân công trọng tài theo vòng đấu}]
CREATE OR REPLACE FUNCTION fn_check_phan_cong_trong_tai()
RETURNS TRIGGER AS $$
DECLARE
    v_ma_vong INT;
    v_trung_vong INT;
BEGIN
    SELECT MaVong INTO v_ma_vong FROM TRAN_DAU WHERE MaTran = NEW.MaTran;
    PERFORM 1 FROM TRONG_TAI WHERE MaTrongTai = NEW.MaTrongTai FOR NO KEY UPDATE;

    SELECT COUNT(*) INTO v_trung_vong
    FROM PHAN_CONG_TRONG_TAI pc
    JOIN TRAN_DAU td ON pc.MaTran = td.MaTran
    WHERE pc.MaTrongTai = NEW.MaTrongTai
      AND td.MaVong = v_ma_vong
      AND td.MaTran <> NEW.MaTran;

    IF v_trung_vong > 0 THEN
        RAISE EXCEPTION 'Loi nghiep vu: Trong tai (%) da dieu khien tran khac trong vong (%)!',
            NEW.MaTrongTai, v_ma_vong;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_check_phan_cong_trong_tai ON PHAN_CONG_TRONG_TAI;
CREATE TRIGGER trg_check_phan_cong_trong_tai
BEFORE INSERT OR UPDATE OF MaTrongTai, MaTran ON PHAN_CONG_TRONG_TAI
FOR EACH ROW EXECUTE FUNCTION fn_check_phan_cong_trong_tai();
\end{lstlisting}

\subsection{Bộ kích hoạt đối soát bàn thắng (kể cả phản lưới) và thẻ phạt}
Khi trận đấu kết thúc, bộ kích hoạt kiểm tra tổ đủ 4 trọng tài, đếm tổng bàn thắng (bao gồm bàn thắng của cầu thủ đội nhà và bàn phản lưới nhà của cầu thủ đội khách) cùng các loại thẻ phạt, đảm bảo trùng khớp $100\%$ với kết quả trận đấu:

\begin{lstlisting}[language=SQL, caption={Bộ kích hoạt đối soát kết quả trận đấu với bảng sự kiện sau chuẩn hóa BCNF}]
CREATE OR REPLACE FUNCTION fn_check_khop_ban_thang_va_the()
RETURNS TRIGGER AS $$
DECLARE
    v_ban_nha INT; v_ban_khach INT;
    v_vang_nha INT; v_vang_khach INT;
    v_do_nha INT; v_do_khach INT;
    v_so_trong_tai INT;
BEGIN
    IF NEW.TrangThai = 'DaKetThuc' THEN
        SELECT COUNT(*) INTO v_so_trong_tai FROM PHAN_CONG_TRONG_TAI WHERE MaTran = NEW.MaTran;
        IF v_so_trong_tai <> 4 THEN
            RAISE EXCEPTION 'Loi nghiep vu: Tran dau (%) phai co du 4 trong tai!', NEW.MaTran;
        END IF;

        -- Tinh ban thang: Ban nha = BanThang cua doi nha + PhanLuoi cua doi khach
        SELECT
            COUNT(*) FILTER (WHERE (dk.MaDoi = NEW.MaDoiNha AND sk.LoaiSuKien = 'BanThang')
                                OR (dk.MaDoi = NEW.MaDoiKhach AND sk.LoaiSuKien = 'PhanLuoi')),
            COUNT(*) FILTER (WHERE (dk.MaDoi = NEW.MaDoiKhach AND sk.LoaiSuKien = 'BanThang')
                                OR (dk.MaDoi = NEW.MaDoiNha AND sk.LoaiSuKien = 'PhanLuoi')),
            COUNT(*) FILTER (WHERE dk.MaDoi = NEW.MaDoiNha AND sk.LoaiSuKien = 'TheVang'),
            COUNT(*) FILTER (WHERE dk.MaDoi = NEW.MaDoiKhach AND sk.LoaiSuKien = 'TheVang'),
            COUNT(*) FILTER (WHERE dk.MaDoi = NEW.MaDoiNha AND sk.LoaiSuKien = 'TheDo'),
            COUNT(*) FILTER (WHERE dk.MaDoi = NEW.MaDoiKhach AND sk.LoaiSuKien = 'TheDo')
        INTO v_ban_nha, v_ban_khach, v_vang_nha, v_vang_khach, v_do_nha, v_do_khach
        FROM SU_KIEN_TRAN_DAU sk
        JOIN DANG_KY_THI_DAU dk ON sk.MaTran = dk.MaTran AND sk.MaCauThu = dk.MaCauThu
        WHERE sk.MaTran = NEW.MaTran;

        IF v_ban_nha <> NEW.BanThangNha OR v_ban_khach <> NEW.BanThangKhach THEN
            RAISE EXCEPTION 'Loi toan ven: So ban thang trong su kien khong khop ti so tran dau!';
        END IF;
        IF v_vang_nha <> NEW.TheVangNha OR v_vang_khach <> NEW.TheVangKhach OR
           v_do_nha <> NEW.TheDoNha OR v_do_khach <> NEW.TheDoKhach THEN
            RAISE EXCEPTION 'Loi toan ven: So the phat trong su kien khong khop thong ke tran dau!';
        END IF;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

DROP TRIGGER IF EXISTS trg_check_khop_ban_thang_va_the ON TRAN_DAU;
CREATE TRIGGER trg_check_khop_ban_thang_va_the
AFTER INSERT OR UPDATE OF TrangThai, BanThangNha, BanThangKhach, TheVangNha, TheVangKhach, TheDoNha, TheDoKhach
ON TRAN_DAU FOR EACH ROW EXECUTE FUNCTION fn_check_khop_ban_thang_va_the();

-- Bo kich hoat ngan sua doi su kien khi tran da DaKetThuc

\subsection{Thủ tục giao dịch tạo lịch thi đấu}
Sử dụng khóa dòng trên sân vận động và so sánh khoảng thời gian \texttt{NgayGio > p\_ngay\_gio - INTERVAL '4 hours' AND NgayGio < p\_ngay\_gio + INTERVAL '4 hours'} để khai thác trực tiếp chỉ mục B-Tree \texttt{idx\_tran\_dau\_san\_ngaygio}:

\begin{lstlisting}[language=SQL, caption={Thủ tục giao dịch lập lịch thi đấu tối ưu hóa chỉ mục}]
CREATE OR REPLACE PROCEDURE sp_tao_lich_thi_dau(
    p_ma_tran VARCHAR(10), p_ma_vong INT,
    p_ma_doi_nha VARCHAR(10), p_ma_doi_khach VARCHAR(10),
    p_ma_san VARCHAR(10), p_ngay_gio TIMESTAMP, p_gia_ve DECIMAL(12, 2)
)
LANGUAGE plpgsql AS $$
DECLARE
    v_da_ton_tai INT; v_trung_lich_san INT;
BEGIN
    IF p_ma_doi_nha = p_ma_doi_khach THEN
        RAISE EXCEPTION 'Loi giao dich: Doi chu nha va doi khach khong duoc trung nhau!';
    END IF;

    PERFORM 1 FROM SAN_VAN_DONG WHERE MaSan = p_ma_san FOR NO KEY UPDATE;

    SELECT COUNT(*) INTO v_da_ton_tai FROM TRAN_DAU
    WHERE MaDoiNha = p_ma_doi_nha AND MaDoiKhach = p_ma_doi_khach;
    IF v_da_ton_tai > 0 THEN
        RAISE EXCEPTION 'Loi giao dich: Cap dau da ton tai tren san nha nay!';
    END IF;

    SELECT COUNT(*) INTO v_trung_lich_san FROM TRAN_DAU
    WHERE MaSan = p_ma_san
      AND NgayGio > p_ngay_gio - INTERVAL '4 hours'
      AND NgayGio < p_ngay_gio + INTERVAL '4 hours';
    IF v_trung_lich_san > 0 THEN
        RAISE EXCEPTION 'Loi giao dich: San da co tran dau trong khoang 4 gio lan can!';
    END IF;

    INSERT INTO TRAN_DAU (
        MaTran, MaVong, MaDoiNha, MaDoiKhach, MaSan, NgayGio, GiaVe,
        SoKhach, BanThangNha, BanThangKhach, TheVangNha, TheVangKhach, TheDoNha, TheDoKhach, TrangThai
    ) VALUES (
        p_ma_tran, p_ma_vong, p_ma_doi_nha, p_ma_doi_khach, p_ma_san, p_ngay_gio, p_gia_ve,
        0, 0, 0, 0, 0, 0, 0, 'ChuaDienRa'
    );
END;
$$;
\end{lstlisting}

\subsection{Thủ tục giao dịch chuyển nhượng hoặc cho mượn cầu thủ}
Cập nhật trạng thái hợp đồng hiện hành (chuyển sang \texttt{'TamDung'} nếu cho mượn để bảo lưu quyền sở hữu của CLB chủ quản, hoặc \texttt{'ThanhLy'} nếu chuyển nhượng dứt đoạn) và khởi tạo hợp đồng mới tại đội bóng tiếp nhận, đảm bảo không bị lỗi trùng khóa nhờ Partial Unique Index:

\begin{lstlisting}[language=SQL, caption={Thủ tục giao dịch chuyển nhượng hoặc cho mượn cầu thủ}]
CREATE OR REPLACE PROCEDURE sp_chuyen_nhuong_cho_muon_cau_thu(
    p_ma_hd_moi VARCHAR(10), p_ma_cau_thu VARCHAR(10),
    p_ma_doi_chu_quan VARCHAR(10), p_ma_doi_moi VARCHAR(10),
    p_so_ao_moi INT, p_ngay_bat_dau DATE, p_ngay_ket_thuc DATE, p_trang_thai VARCHAR(20)
)
LANGUAGE plpgsql AS $$
BEGIN
    PERFORM 1 FROM CAU_THU WHERE MaCauThu = p_ma_cau_thu FOR NO KEY UPDATE;
    PERFORM 1 FROM DOI_BONG WHERE MaDoi = p_ma_doi_moi FOR NO KEY UPDATE;

    -- Xu ly hop dong cu dang active:
    -- Neu cho muon (ChoMuon): Hop dong goc chuyen sang 'TamDung' de bao luu quyen so huu cua CLB chu quan
    -- Neu chuyen nhuong dut doan (ChinhThuc): Hop dong cu chuyen sang 'ThanhLy'
    IF p_trang_thai = 'ChoMuon' THEN
        UPDATE HOP_DONG_CAU_THU
        SET TrangThai = 'TamDung'
        WHERE MaCauThu = p_ma_cau_thu AND TrangThai IN ('ChinhThuc', 'ChoMuon')
          AND NgayKetThuc >= p_ngay_bat_dau;
    ELSE
        UPDATE HOP_DONG_CAU_THU
        SET NgayKetThuc = p_ngay_bat_dau - INTERVAL '1 day', TrangThai = 'ThanhLy'
        WHERE MaCauThu = p_ma_cau_thu AND TrangThai IN ('ChinhThuc', 'ChoMuon', 'TamDung')
          AND NgayKetThuc >= p_ngay_bat_dau;
    END IF;

    -- Tao hop dong moi (Partial Unique Index chi kiem tra cac hop dong dang hieu luc thi dau)
    INSERT INTO HOP_DONG_CAU_THU (
        MaHD, MaCauThu, MaDoiChuQuan, MaDoiHienTai, SoAo, NgayBatDau, NgayKetThuc, TrangThai
    ) VALUES (
        p_ma_hd_moi, p_ma_cau_thu, p_ma_doi_chu_quan, p_ma_doi_moi,
        p_so_ao_moi, p_ngay_bat_dau, p_ngay_ket_thuc, p_trang_thai
    );
END;
$$;
\end{lstlisting}

\section{Cài đặt các khung nhìn và hàm nghiệp vụ thống kê}

\subsection{Khung nhìn bảng xếp hạng mùa giải}
Bảng xếp hạng được tính toán tự động từ tất cả các trận đấu đã kết thúc, sắp xếp theo đúng 6 tiêu chí ưu tiên của điều lệ giải:

\begin{lstlisting}[language=SQL, caption={Khung nhìn bảng xếp hạng mùa giải}]
CREATE OR REPLACE VIEW vw_bang_xep_hang AS
WITH TranDauKetThuc AS (
    SELECT MaDoiNha AS MaDoi, 1 AS SoTran,
        CASE WHEN BanThangNha > BanThangKhach THEN 1 ELSE 0 END AS Thang,
        CASE WHEN BanThangNha = BanThangKhach THEN 1 ELSE 0 END AS Hoa,
        CASE WHEN BanThangNha < BanThangKhach THEN 1 ELSE 0 END AS Thua,
        BanThangNha AS BanThang, BanThangKhach AS BanThua,
        0 AS ThangSanKhach, TheDoNha AS TheDo, TheVangNha AS TheVang
    FROM TRAN_DAU WHERE TrangThai = 'DaKetThuc'
    UNION ALL
    SELECT MaDoiKhach AS MaDoi, 1 AS SoTran,
        CASE WHEN BanThangKhach > BanThangNha THEN 1 ELSE 0 END AS Thang,
        CASE WHEN BanThangKhach = BanThangNha THEN 1 ELSE 0 END AS Hoa,
        CASE WHEN BanThangKhach < BanThangNha THEN 1 ELSE 0 END AS Thua,
        BanThangKhach AS BanThang, BanThangNha AS BanThua,
        CASE WHEN BanThangKhach > BanThangNha THEN 1 ELSE 0 END AS ThangSanKhach,
        TheDoKhach AS TheDo, TheVangKhach AS TheVang
    FROM TRAN_DAU WHERE TrangThai = 'DaKetThuc'
),
TongHop AS (
    SELECT d.MaDoi, d.TenDoi,
        COALESCE(SUM(t.SoTran), 0) AS SoTranDaDau,
        COALESCE(SUM(t.Thang), 0) AS SoTranThang,
        COALESCE(SUM(t.Hoa), 0) AS SoTranHoa,
        COALESCE(SUM(t.Thua), 0) AS SoTranThua,
        COALESCE(SUM(t.BanThang), 0) AS TongBanThang,
        COALESCE(SUM(t.BanThua), 0) AS TongBanThua,
        COALESCE(SUM(t.BanThang) - SUM(t.BanThua), 0) AS HieuSo,
        COALESCE(SUM(t.Thang) * 3 + SUM(t.Hoa), 0) AS DiemSo,
        COALESCE(SUM(t.ThangSanKhach), 0) AS SoTranThangSanKhach,
        COALESCE(SUM(t.TheDo), 0) AS TongTheDo,
        COALESCE(SUM(t.TheVang), 0) AS TongTheVang
    FROM DOI_BONG d LEFT JOIN TranDauKetThuc t ON d.MaDoi = t.MaDoi
    GROUP BY d.MaDoi, d.TenDoi
)
SELECT ROW_NUMBER() OVER (
    ORDER BY DiemSo DESC, HieuSo DESC, TongBanThang DESC,
             SoTranThangSanKhach DESC, TongTheDo ASC, TongTheVang ASC, TenDoi ASC
) AS ViTri, * FROM TongHop;
\end{lstlisting}

\subsection{Hàm thống kê Vua phá lưới và tính thù lao trọng tài theo tháng}
Hàm \texttt{fn\_top10\_vua\_pha\_luoi} gom nhóm theo từng cầu thủ để bảo đảm nếu cầu thủ chuyển nhượng giữa mùa thì tổng bàn thắng vẫn được cộng dồn chính xác; đội bóng hiện tại được lấy từ hợp đồng mới nhất còn hiệu lực:

\begin{lstlisting}[language=SQL, caption={Hàm truy vấn Vua phá lưới và tính thù lao trọng tài theo tháng}]
CREATE OR REPLACE FUNCTION fn_top10_vua_pha_luoi()
RETURNS TABLE (
    XepHang BIGINT, MaCauThu VARCHAR(10), HoTen VARCHAR(100),
    TenDoiHienTai VARCHAR(100), SoBanThang BIGINT
) LANGUAGE sql AS $$
    WITH BanThangCuaCauThu AS (
        SELECT sk.MaCauThu, COUNT(*) AS SoBanThang
        FROM SU_KIEN_TRAN_DAU sk WHERE sk.LoaiSuKien = 'BanThang'
        GROUP BY sk.MaCauThu
    ),
    DoiBongHienTai AS (
        SELECT DISTINCT ON (hd.MaCauThu) hd.MaCauThu, db.TenDoi
        FROM HOP_DONG_CAU_THU hd
        JOIN DOI_BONG db ON hd.MaDoiHienTai = db.MaDoi
        WHERE hd.TrangThai IN ('ChinhThuc', 'ChoMuon')
        ORDER BY hd.MaCauThu, hd.NgayBatDau DESC
    )
    SELECT ROW_NUMBER() OVER (ORDER BY b.SoBanThang DESC, ct.HoTen ASC) AS XepHang,
           ct.MaCauThu, ct.HoTen, COALESCE(d.TenDoi, 'Tu do') AS TenDoiHienTai, b.SoBanThang
    FROM BanThangCuaCauThu b
    JOIN CAU_THU ct ON b.MaCauThu = ct.MaCauThu
    LEFT JOIN DoiBongHienTai d ON b.MaCauThu = d.MaCauThu
    ORDER BY b.SoBanThang DESC, ct.HoTen ASC LIMIT 10;
$$;

CREATE OR REPLACE FUNCTION fn_tinh_luong_trong_tai_thang(p_thang INT, p_nam INT)
RETURNS TABLE (
    MaTrongTai VARCHAR(10), HoTen VARCHAR(100), CapBac VARCHAR(50),
    SoTranChinh BIGINT, SoTranBien BIGINT, SoTranBan BIGINT,
    TongSoTran BIGINT, TongLuong DECIMAL(15, 2)
) LANGUAGE sql AS $$
    SELECT tt.MaTrongTai, tt.HoTen, tt.CapBac,
        COUNT(CASE WHEN td.MaTran IS NOT NULL AND pc.VaiTro = 'Chinh' THEN 1 END),
        COUNT(CASE WHEN td.MaTran IS NOT NULL AND pc.VaiTro IN ('Bien1', 'Bien2') THEN 1 END),
        COUNT(CASE WHEN td.MaTran IS NOT NULL AND pc.VaiTro = 'Ban' THEN 1 END),
        COUNT(td.MaTran),
        (COUNT(CASE WHEN td.MaTran IS NOT NULL AND pc.VaiTro = 'Chinh' THEN 1 END) * 3000000 +
         COUNT(CASE WHEN td.MaTran IS NOT NULL AND pc.VaiTro IN ('Bien1', 'Bien2') THEN 1 END) * 2000000 +
         COUNT(CASE WHEN td.MaTran IS NOT NULL AND pc.VaiTro = 'Ban' THEN 1 END) * 1000000)::DECIMAL(15, 2)
    FROM TRONG_TAI tt
    LEFT JOIN (
        PHAN_CONG_TRONG_TAI pc
        JOIN TRAN_DAU td ON pc.MaTran = td.MaTran
                        AND EXTRACT(MONTH FROM td.NgayGio) = p_thang
                        AND EXTRACT(YEAR FROM td.NgayGio) = p_nam
                        AND td.TrangThai = 'DaKetThuc'
    ) ON tt.MaTrongTai = pc.MaTrongTai
    GROUP BY tt.MaTrongTai, tt.HoTen, tt.CapBac
    ORDER BY 8 DESC, tt.HoTen ASC;
$$;
\end{lstlisting}

\subsection{Hàm thống kê doanh thu sân vận động và diễn biến vòng đấu}
Đáp ứng đầy đủ các yêu cầu đã cam kết ở mục 1.1.3:

\begin{lstlisting}[language=SQL, caption={Hàm thống kê doanh thu bán vé và diễn biến chi tiết vòng đấu}]
CREATE OR REPLACE FUNCTION fn_thong_ke_doanh_thu_san()
RETURNS TABLE (
    MaSan VARCHAR(10), TenSan VARCHAR(100), DiaChi VARCHAR(200),
    SucChua INT, SoTranDaToChuc BIGINT, TongKhanGia BIGINT, TongDoanhThu DECIMAL(18, 2)
) LANGUAGE sql AS $$
    SELECT s.MaSan, s.TenSan, s.DiaChi, s.SucChua,
        COUNT(t.MaTran) AS SoTranDaToChuc,
        COALESCE(SUM(t.SoKhach), 0)::BIGINT AS TongKhanGia,
        COALESCE(SUM(t.SoKhach * t.GiaVe), 0)::DECIMAL(18, 2) AS TongDoanhThu
    FROM SAN_VAN_DONG s
    LEFT JOIN TRAN_DAU t ON s.MaSan = t.MaSan AND t.TrangThai = 'DaKetThuc'
    GROUP BY s.MaSan, s.TenSan, s.DiaChi, s.SucChua
    ORDER BY TongDoanhThu DESC, s.TenSan ASC;
$$;

CREATE OR REPLACE FUNCTION fn_thong_ke_vong_dau(p_ma_vong INT)
RETURNS TABLE (
    MaTran VARCHAR(10), NgayGio TIMESTAMP, TenSan VARCHAR(100),
    DoiNha VARCHAR(100), DoiKhach VARCHAR(100), TiSo VARCHAR(20),
    TheVang INT, TheDo INT, SoKhach INT, TrangThai VARCHAR(20)
) LANGUAGE sql AS $$
    SELECT t.MaTran, t.NgayGio, s.TenSan, dn.TenDoi AS DoiNha, dk.TenDoi AS DoiKhach,
        CONCAT(t.BanThangNha, ' - ', t.BanThangKhach) AS TiSo,
        (t.TheVangNha + t.TheVangKhach) AS TheVang, (t.TheDoNha + t.TheDoKhach) AS TheDo,
        t.SoKhach, t.TrangThai
    FROM TRAN_DAU t
    JOIN DOI_BONG dn ON t.MaDoiNha = dn.MaDoi
    JOIN DOI_BONG dk ON t.MaDoiKhach = dk.MaDoi
    JOIN SAN_VAN_DONG s ON t.MaSan = s.MaSan
    WHERE t.MaVong = p_ma_vong ORDER BY t.NgayGio ASC;
$$;
\end{lstlisting}

\section{Kết quả nạp dữ liệu và kiểm thử thực nghiệm}

Hệ thống đã nạp trọn vẹn dữ liệu mẫu thực tế của mùa giải gồm 6 đội bóng, 108 cầu thủ, 12 trọng tài, 30 trận đấu (10 vòng đấu), 120 bản ghi phân công trọng tài, 1.050 bản ghi đăng ký cầu thủ thi đấu và 166 sự kiện bàn thắng, thẻ phạt.

\textit{Giải trình số liệu 1.050 lượt đăng ký thi đấu}: Về mặt lý thuyết, trong 30 trận đấu ($30 \times 2 = 60$ lượt đội ra sân) với quy định tối đa 18 cầu thủ/đội, tổng số lượt đăng ký trần có thể đạt $60 \times 18 = 1.080$ lượt. Tuy nhiên trên thực tế, các đội bóng không đăng ký tối đa đủ 18 cầu thủ ở một số trận do có các cầu thủ phải thụ án kỷ luật treo giò (từ 2 thẻ đỏ và các trường hợp tích lũy đủ 2 thẻ vàng) hoặc chấn thương/xoay tua đội hình của ban huấn luyện, dẫn đến 30 lượt đội đăng ký 18 cầu thủ và 30 lượt đội đăng ký 17 cầu thủ, tạo nên đúng tổng số $30 \times 18 + 30 \times 17 = 540 + 510 = 1.050$ bản ghi đăng ký thi đấu, phản ánh chính xác tính thực tế của giải đấu chuyên nghiệp.

\textbf{Kiểm chứng các bất biến toán học trên bảng xếp hạng:}
\begin{itemize}
    \item \textbf{Cân bằng Thắng -- Thua}: Tổng số trận thắng của 6 đội là $8 + 6 + 4 + 3 + 1 + 1 = 23$ trận; tổng số trận thua là $0 + 1 + 3 + 3 + 8 + 8 = 23$ trận. Ta có $\sum \text{Thắng} \equiv \sum \text{Thua} = 23$.
    \item \textbf{Tổng số trận đấu}: Tổng lượt trận hòa của các đội là $2 + 3 + 3 + 4 + 1 + 1 = 14$ lượt, tương ứng với $14 / 2 = 7$ trận hòa. Tổng số trận của mùa giải: $23 \text{ (thắng/thua)} + 7 \text{ (hòa)} = 30$ trận đấu, hoàn toàn khớp với kế hoạch thi đấu vòng tròn 2 lượt của 6 đội.
    \item \textbf{Cân bằng Bàn thắng -- Bàn thua}: Tổng bàn thắng ghi được toàn giải: $21 + 18 + 16 + 13 + 8 + 5 = 81$ bàn (bao gồm cả bàn thắng trực tiếp và bàn phản lưới nhà); tổng bàn thua: $8 + 11 + 11 + 11 + 18 + 22 = 81$ bàn. Hiệu số bàn thắng thua toàn giải triệt tiêu về $0$.
    \item \textbf{Kiểm chứng thẻ phạt và tổng số sự kiện}: Tổng số thẻ phạt trên bảng xếp hạng là 83 thẻ vàng và 2 thẻ đỏ, khớp chính xác $100\%$ với các sự kiện thẻ phạt ghi nhận trong bảng \texttt{SU\_KIEN\_TRAN\_DAU}. Toàn bộ mùa giải ghi nhận đúng $81 \text{ bàn thắng} + 83 \text{ thẻ vàng} + 2 \text{ thẻ đỏ} = 166 \text{ sự kiện diễn biến trận đấu}$, hoàn toàn đồng nhất trên toàn hệ thống.
\end{itemize}

\chapter{KHẢO SÁT BÀI TOÁN VÀ THIẾT KẾ LƯỢC ĐỒ THỰC THỂ LIÊN KẾT}

\section{Mô tả bài toán và khảo sát yêu cầu}

\subsection{Bối cảnh và quy mô hệ thống}
Hệ thống cơ sở dữ liệu quản lý giải bóng đá chuyên nghiệp được xây dựng nhằm phục vụ công tác tổ chức, điều hành và theo dõi toàn diện các hoạt động trong suốt một mùa giải. Hệ thống quản lý thông tin các câu lạc bộ, cầu thủ, sân vận động, đội ngũ trọng tài, lập lịch thi đấu vòng tròn hai lượt, tiếp nhận danh sách đăng ký thi đấu trước mỗi trận, ghi nhận chi tiết các sự kiện diễn ra trên sân và tự động tổng hợp bảng xếp hạng sau mỗi vòng đấu.

Các thông số quy định của giải đấu bao gồm:
\begin{itemize}
    \item \textbf{Số lượng đội bóng tham gia}: Giải đấu gồm $N = 6$ đội bóng (đáp ứng đúng tiêu chí số chẵn từ 6 đội trở lên).
    \item \textbf{Thể thức thi đấu}: Thi đấu vòng tròn 2 lượt gồm lượt đi trên sân nhà và lượt về trên sân khách. Với 6 đội tham gia, mùa giải có tất cả $2 \times (6 - 1) = 10$ vòng đấu. Mỗi vòng đấu diễn ra 3 trận đấu giữa 3 cặp đối đầu.
    \item \textbf{Tổng số trận đấu trong cả mùa giải}: Toàn mùa giải gồm $10 \text{ vòng} \times 3 \text{ trận/vòng} = 30 \text{ trận đấu}$. Lịch thi đấu và kết quả của toàn bộ 30 trận đấu này được lưu trữ đầy đủ trong cơ sở dữ liệu.
\end{itemize}

\subsection{Các quy tắc nghiệp vụ}
Hệ thống tuân thủ chặt chẽ các quy định nghiệp vụ sau:
\begin{enumerate}
    \item \textbf{Quy định về đội bóng, sân nhà và trang phục}:
    \begin{itemize}
        \item Mỗi đội bóng có đúng một sân nhà cố định. Một sân vận động có thể được đăng ký làm sân nhà của một hoặc nhiều đội bóng khác nhau.
        \item Mỗi đội bóng phải đăng ký từ hai màu áo thi đấu trở lên gồm màu áo sân nhà, màu áo sân khách và áo phụ.
        \item Một đội bóng được đăng ký tối đa 30 cầu thủ trong danh sách thi đấu của mùa giải.
    \end{itemize}
    \item \textbf{Quy định về cầu thủ, hợp đồng và số áo}:
    \begin{itemize}
        \item Mỗi cầu thủ ký hợp đồng chuyên nghiệp với một đội bóng chủ quản. Cầu thủ có thể chuyển sang thi đấu cho một đội bóng khác theo dạng cho mượn.
        \item Tại mỗi thời điểm, mỗi cầu thủ chỉ được thi đấu cho đúng một đội bóng.
        \item Số áo thi đấu của cầu thủ nằm trong khoảng từ $1$ đến $35$.
        \item Trong cùng một đội bóng, các cầu thủ phải có số áo khác nhau; các cầu thủ thuộc hai đội bóng khác nhau có thể mang cùng số áo.
    \end{itemize}
    \item \textbf{Quy định về trọng tài và điều hành trận đấu}:
    \begin{itemize}
        \item Mỗi trận đấu có đúng bốn trọng tài làm nhiệm vụ gồm: một trọng tài chính, hai trọng tài biên và một trọng tài bàn.
        \item Trong cùng một vòng đấu, một trọng tài chỉ được điều khiển tối đa một trận đấu.
    \end{itemize}
    \item \textbf{Quy định về đăng ký thi đấu và kết quả trận đấu}:
    \begin{itemize}
        \item Trận đấu có hai trạng thái: chưa diễn ra hoặc đã kết thúc. Ngày diễn ra trận đấu không được vượt quá thời gian hiện tại đối với các trận đã kết thúc.
        \item Trong mỗi trận đấu, mỗi đội bóng chỉ được đăng ký tối đa 18 cầu thủ gồm 11 cầu thủ đá chính và tối đa 7 cầu thủ dự bị ($11 + 7 = 18$).
        \item Thông tin về bàn thắng, thẻ vàng, thẻ đỏ mà từng cầu thủ ghi được hoặc mắc phải trong mỗi trận đấu đều được ghi nhận chi tiết theo từng phút.
        \item Tổng số bàn thắng và tổng số thẻ phạt của các cầu thủ trong đội phải khớp chính xác với tỉ số và số thẻ ghi nhận của trận đấu đó.
    \end{itemize}
\end{enumerate}

\subsection{Các chức năng triển khai trên hệ cơ sở dữ liệu}
Hệ thống cơ sở dữ liệu đáp ứng các chức năng nghiệp vụ sau:
\begin{itemize}
    \item \textbf{Quản lý dữ liệu danh mục}: Thêm, sửa, xóa, tra cứu thông tin đội bóng, cầu thủ, trọng tài, sân vận động, hợp đồng chuyển nhượng và cho mượn cầu thủ.
    \item \textbf{Bảng xếp hạng giải đấu}: Tự động tính điểm (thắng được 3 điểm, hòa được 1 điểm, thua được 0 điểm) và xếp hạng các đội lần lượt theo các tiêu chí:
    \begin{align*}
        \text{Điểm số} &\rightarrow \text{Hiệu số bàn thắng/thua} \rightarrow \text{Tổng số bàn thắng} \\
        &\rightarrow \text{Số trận thắng sân khách} \rightarrow \text{Ít thẻ đỏ hơn} \rightarrow \text{Ít thẻ vàng hơn}.
    \end{align*}
    \item \textbf{Thống kê vua phá lưới}: Hiển thị danh sách 10 cầu thủ ghi được nhiều bàn thắng nhất trong mùa giải.
    \item \textbf{Tính thù lao trọng tài theo tháng}: Tính tiền công điều khiển trận đấu cho các trọng tài theo quy định: trọng tài chính nhận $3.000.000$ đồng/trận; trọng tài biên nhận $2.000.000$ đồng/trận; trọng tài bàn nhận $1.000.000$ đồng/trận.
    \item \textbf{Thống kê doanh thu sân vận động}: Tính toán doanh thu bán vé từng sân đấu theo công thức: $\text{Doanh thu} = \sum (\text{Số khán giả} \times \text{Giá vé mỗi trận})$.
    \item \textbf{Thống kê chi tiết vòng đấu}: Tra cứu kết quả, lịch thi đấu, tỉ số và các sự kiện diễn ra của một vòng đấu bất kỳ.
\end{itemize}

\section{Thiết kế Lược đồ Thực thể Liên kết}

\subsection{Xác định các tập thực thể và thuộc tính}
Hệ thống gồm các tập thực thể và mối kết hợp sau:
\begin{itemize}
    \item \textbf{Tập thực thể mạnh}:
    \begin{itemize}
        \item \texttt{SAN\_VAN\_DONG}: Khóa chính là \underline{\texttt{MaSan}}.
        \item \texttt{DOI\_BONG}: Khóa chính là \underline{\texttt{MaDoi}}.
        \item \texttt{CAU\_THU}: Khóa chính là \underline{\texttt{MaCauThu}}.
        \item \texttt{TRONG\_TAI}: Khóa chính là \underline{\texttt{MaTrongTai}}.
        \item \texttt{VONG\_DAU}: Khóa chính là \underline{\texttt{MaVong}}.
    \end{itemize}
    \item \textbf{Tập thực thể yếu, thực thể giao dịch và mối kết hợp đa trị}:
    \begin{itemize}
        \item \texttt{MAU\_AO\_DOI\_BONG}: Thực thể yếu phụ thuộc vào \texttt{DOI\_BONG}, khóa bộ phận là \underline{\texttt{LoaiMauAo}}.
        \item \texttt{HOP\_DONG\_CAU\_THU}: Thực thể giao dịch biểu diễn hợp đồng chuyên nghiệp và trạng thái cho mượn giữa cầu thủ với đội bóng chủ quản và đội bóng hiện tại, khóa chính là \underline{\texttt{MaHD}}.
        \item \texttt{TRAN\_DAU}: Thực thể giao dịch biểu diễn cuộc đối đầu giữa hai đội bóng trong một vòng đấu tại một sân vận động cụ thể, khóa chính là \underline{\texttt{MaTran}}.
        \item \texttt{PHAN\_CONG\_TRONG\_TAI}: Mối kết hợp nhiều -- nhiều giữa \texttt{TRAN\_DAU} và \texttt{TRONG\_TAI}, khóa chính tổ hợp gồm (\underline{\texttt{MaTran}}, \underline{\texttt{VaiTro}}).
        \item \texttt{DANG\_KY\_THI\_DAU}: Thực thể kết hợp biểu diễn danh sách đăng ký cầu thủ của đội bóng trong một trận đấu, khóa chính tổ hợp gồm (\underline{\texttt{MaTran}}, \underline{\texttt{MaCauThu}}).
        \item \texttt{SU\_KIEN\_TRAN\_DAU}: Thực thể phụ thuộc vào \texttt{DANG\_KY\_THI\_DAU}, ghi nhận các sự kiện bàn thắng và thẻ phạt phát sinh của cầu thủ đã đăng ký trong trận, khóa chính là \underline{\texttt{MaSuKien}}.
    \end{itemize}
\end{itemize}

\subsection{Các mối quan hệ và bản số}
\begin{itemize}
    \item \textbf{Mối kết hợp \textit{Sân nhà} (\texttt{SAN\_VAN\_DONG} -- \texttt{DOI\_BONG})}:
    Quan hệ $1 - N$. Một sân vận động có thể là sân nhà của không, một hoặc nhiều đội bóng với bản số $(0, N)$. Mỗi đội bóng bắt buộc đăng ký đúng một sân nhà cố định với bản số $(1, 1)$.
    \item \textbf{Mối kết hợp \textit{Màu áo} (\texttt{DOI\_BONG} -- \texttt{MAU\_AO\_DOI\_BONG})}:
    Quan hệ $1 - N$. Một đội bóng phải đăng ký từ 2 đến 3 bộ màu áo thi đấu với bản số $(2, 3)$. Mỗi bản ghi màu áo thuộc về duy nhất một đội bóng với bản số $(1, 1)$.
    \item \textbf{Mối kết hợp \textit{Hợp đồng cầu thủ} (\texttt{CAU\_THU} -- \texttt{HOP\_DONG\_CAU\_THU} -- \texttt{DOI\_BONG})}:
    Quan hệ $1 - N$ từ \texttt{CAU\_THU} và \texttt{DOI\_BONG} tới \texttt{HOP\_DONG\_CAU\_THU}. Mỗi cầu thủ trong mùa giải có bản số $(1, N)$ theo lịch sử hợp đồng nhưng tại một thời điểm chỉ có duy nhất một hợp đồng hiệu lực. Mỗi đội bóng đăng ký danh sách cầu thủ trong mùa giải với bản số $(1, N)$ (theo quy định điều lệ giải, mỗi đội có tối thiểu 18 và tối đa 30 cầu thủ có hợp đồng hiệu lực).
    \item \textbf{Mối kết hợp \textit{Lịch thi đấu theo vòng} (\texttt{VONG\_DAU} -- \texttt{TRAN\_DAU})}:
    Quan hệ $1 - N$. Một vòng đấu có thể có một hoặc nhiều trận đấu với bản số tổng quát $(1, N)$ (cụ thể trong giải 6 đội gồm 3 trận/vòng). Mỗi trận đấu thuộc về duy nhất một vòng đấu với bản số $(1, 1)$.
    \item \textbf{Mối kết hợp \textit{Chủ nhà / Đội khách} (\texttt{DOI\_BONG} -- \texttt{TRAN\_DAU})}:
    Hai mối kết hợp $1 - N$ độc lập ứng với vai trò đội chủ nhà và đội khách. Một đội bóng có thể tham gia nhiều trận đấu với vai trò đội chủ nhà $(0, N)$ và nhiều trận đấu với vai trò đội khách $(0, N)$; mỗi trận đấu có đúng một đội chủ nhà $(1, 1)$ và một đội khách $(1, 1)$.
    \item \textbf{Mối kết hợp \textit{Tổ chức tại sân} (\texttt{SAN\_VAN\_DONG} -- \texttt{TRAN\_DAU})}:
    Quan hệ $1 - N$. Một sân vận động có thể tổ chức nhiều trận đấu trong mùa giải với bản số $(0, N)$; mỗi trận đấu diễn ra tại đúng một sân vận động thực tế với bản số $(1, 1)$.
    \item \textbf{Mối kết hợp \textit{Phân công điều khiển} (\texttt{TRAN\_DAU} -- \texttt{TRONG\_TAI})}:
    Quan hệ $N - N$ được triển khai qua thực thể kết hợp \texttt{PHAN\_CONG\_TRONG\_TAI}. Mỗi trận đấu có nhiều trọng tài được phân công điều hành với bản số $(1, N)$ (cụ thể điều lệ yêu cầu đủ 4 trọng tài); mỗi trọng tài có thể tham gia điều hành nhiều trận đấu trong mùa giải với bản số $(0, N)$ và tối đa 1 trận trong mỗi vòng đấu.
    \item \textbf{Mối kết hợp \textit{Đăng ký danh sách thi đấu} (\texttt{TRAN\_DAU} -- \texttt{CAU\_THU})}:
    Quan hệ $N - N$ được triển khai thành thực thể kết hợp \texttt{DANG\_KY\_THI\_DAU}. Mỗi trận đấu có nhiều cầu thủ được hai đội đăng ký với bản số $(1, N)$ (cụ thể mỗi đội từ 11 đến 18 cầu thủ, gồm 11 cầu thủ đá chính và tối đa 7 cầu thủ dự bị). Mỗi cầu thủ có thể được đăng ký trong nhiều trận đấu suốt mùa giải với bản số $(0, N)$.
    \item \textbf{Mối kết hợp \textit{Ghi nhận sự kiện} (\texttt{DANG\_KY\_THI\_DAU} -- \texttt{SU\_KIEN\_TRAN\_DAU})}:
    Quan hệ $1 - N$. Một cầu thủ được đăng ký trong trận đấu có thể phát sinh từ $0$ đến $N$ sự kiện bàn thắng hoặc thẻ phạt với bản số $(0, N)$; mỗi sự kiện diễn biến thuộc về đúng một cầu thủ đã đăng ký trong trận đấu đó với bản số $(1, 1)$.
\end{itemize}
